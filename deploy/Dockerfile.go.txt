# Go distroless for go.patriotpest.pro — Vanguard: distroless:nonroot
FROM golang:1.22-alpine AS build
WORKDIR /src
COPY go.mod go.sum ./
RUN go mod download
COPY . .
RUN CGO_ENABLED=0 GOOS=linux go build -o /out/patriot-server ./cmd/server

FROM gcr.io/distroless/static-debian12:nonroot
COPY --from=build /out/patriot-server /patriot-server
COPY --from=build /src/migrations /migrations
COPY --from=build /src/configs /configs
EXPOSE 3000
HEALTHCHECK --interval=30s --timeout=3s --start-period=5s --retries=3 CMD ["/patriot-server","-health"]
ENTRYPOINT ["/patriot-server"]
